# Create a function to simulate a fight
num_simulations <- 1000
# for the renewability/flexibility of the code I will attach the pokemons first
#but the function will stay as an universal entity, so as to can be used to other pok
emons
bulbasaur_hp <- 45 # a
pikachu_hp <- 35 # b
bulbasaur_attack <- 49
pikachu_attack <- 55
# In R, we define a function using the function() command and curly braces {}
#First we need to set a general function
simulate_single <- function(hp_a,hp_b,attack_a,attack_b) { #assigning the parameters
  current_hp_a <- hp_a
  current_hp_b <- hp_b
  while(TRUE) {  #Controls the loop/set infinite loop/return() will stop it
    # Pokemon A attacks Pokemon B
    damage_to_b <- attack_a * runif(1, 0.2, 0.6)  #runif() for generating random       
value
    current_hp_b <- current_hp_b - damage_to_b
    if (current_hp_b <= 0) {
      return("Bulbasaur") #return will give the output
    }
    # Pokemon B attacks Pokemon A
    damage_to_a <- attack_b * runif(1, 0.2, 0.6)
    current_hp_a <- current_hp_a - damage_to_a
    if (current_hp_a <= 0) {
      return("Pikachu")
    }
  }
}
#Now we have to run the simulations for 1000 times with using for loop
results <- c()  #First we have to set empty vector to store results, because it will 
overuse memory if we do not preallocate
for (i in 1:1000) { # 1 ve 1000 arası tüm i değerleri için for döngüsü çalışacak
  results[i] <- simulate_single ( # [i], assign a defined variable for ith index of e
ach i value
    hp_a = bulbasaur_hp,
    hp_b = pikachu_hp,
    attack_a = bulbasaur_attack,
    attack_b = pikachu_attack
  )
}
# Now we can analyze the probabilities
#We can both use sum()/1000 and mean() functions to find the probabilities
prob_bulbasaur_wins <- sum(results == "Bulbasaur") /1000
prob_pikachu_wins <- mean(results == "Pikachu")
