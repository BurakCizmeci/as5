R Code Questions 1 & 2: Reshaping & Frequency Table
▼ 
# Select type_1 and type_2 columns and filter out NA values
most_common <- pokemon %>%
  select(type_1, type_2) %>%
  filter(!is.na(type_1) & !is.na(type_2))

# pivot_longer to convert wide format to long format
most_common_long <- most_common %>% 
  pivot_longer(cols = c(type_1, type_2),
               names_to = "type_name",
               values_to = "type_value")

# Count occurrences sorted descending
value <- most_common_long %>%
  count(type_value, sort = TRUE)

# Peak type (Flying: 123)
max_value <- value %>% slice(1)

Analytical Solution to Question 10

Even though Pikachu holds a higher attack stat (55 vs. 49), Bulbasaur's superior HP pool (45 vs. 35) provides decisive durability. With round damage averaging $Attack \times 0.4$, Bulbasaur comfortably survives 2 attacks, while Pikachu frequently faints within round 2, generating an empirical win rate of 85.5% across 1,000 iterations.

R Code Question 10: Original R Simulation Loop
▼ 
simulate_single <- function(hp_a, hp_b, attack_a, attack_b) {
  current_hp_a <- hp_a
  current_hp_b <- hp_b
  while(TRUE) {
    # Pokémon A attacks Pokémon B with uniform random multiplier
    damage_to_b <- attack_a * runif(1, 0.2, 0.6)
    current_hp_b <- current_hp_b - damage_to_b
    if (current_hp_b <= 0) return("Bulbasaur")

    # Pokémon B attacks Pokémon A
    damage_to_a <- attack_b * runif(1, 0.2, 0.6)
    current_hp_a <- current_hp_a - damage_to_a
    if (current_hp_a <= 0) return("Pikachu")
  }
}

# Monte Carlo loop (1,000 iterations)
results <- c()
for (i in 1:1000) {
  results[i] <- simulate_single(
    hp_a = 45, hp_b = 35, attack_a = 49, attack_b = 55
  )
}
prob_bulbasaur_wins <- sum(results == "Bulbasaur") / 1000 # [1] 0.855
